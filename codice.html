Struttura del Codice

Codice in C implementato all'interno dell'Arduino per la gestione dell'accensione e spegnimento dei led.

CODICE ARDUINO

const int lblue = 8;
const int lrr = 9;
const int lbianco = 7;
const int lross = 10;

void setup() {
  pinMode(lblue, OUTPUT);
  pinMode(lrr, OUTPUT);
  pinMode(lbianco, OUTPUT);
  pinMode(lross, OUTPUT);
}

void loop() {

  digitalWrite(lrr, HIGH);
  digitalWrite(lross, HIGH);

  digitalWrite(lblue, LOW);
  digitalWrite(lbianco, LOW);

  delay(2000);

  digitalWrite(lrr, LOW);
  digitalWrite(lross, LOW);
  digitalWrite(lblue, LOW);
  digitalWrite(lbianco, LOW);

  delay(1500);

  digitalWrite(lblue, HIGH);
  digitalWrite(lbianco, HIGH);

  delay(2000);

  digitalWrite(lblue, LOW);
  digitalWrite(lbianco, LOW);

  delay(1500);
}
